```rust
pub(crate) fn unique_target(dir: &Path, file_name: &str, overwrite: bool, used: &HashSet<String>) -> PathBuf {
    // `used` guards against two inputs of the same run mapping to one target:
    // overwrite may replace previous runs' files, never a sibling's result.
    // Keys are lowercased because Windows paths are case-insensitive.
    let is_free = |path: &Path| !used.contains(&target_key(path)) && (overwrite || !path.exists());
    let target = dir.join(file_name);
    if is_free(&target) {
        return target;
    }
    let (stem, ext) = match file_name.rfind('.') {
        Some(pos) => (&file_name[..pos], &file_name[pos..]),
        None => (file_name, ""),
    };
    let mut n = 2;
    loop {
        let candidate = dir.join(format!("{stem} ({n}){ext}"));
        if is_free(&candidate) {
            return candidate;
        }
        n += 1;
    }
}
```

```rust
use crate::domain::{BookHeading, SplitConfig};

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BookPart {
    pub title: Option<String>,
    pub text: String,
}

struct Boundary {
    offset: usize,
    title: Option<String>,
    rank: u8,
}

pub fn validate_config(config: &SplitConfig) -> Result<(), String> {
    if !(config.by_size || config.by_keywords || config.by_blank_lines || config.by_all_caps || config.by_toc) {
        return Err("Включите хотя бы один способ деления".to_string());
    }
    if config.by_keywords && parse_keywords(&config.keywords).is_empty() {
        return Err("Введите одно или несколько ключевых слов для поиска".to_string());
    }
    Ok(())
}

pub fn split_book(text: &str, headings: &[BookHeading], config: &SplitConfig) -> Vec<BookPart> {
    if text.trim().is_empty() {
        return Vec::new();
    }
    let boundaries = collect_boundaries(text, headings, config);
    form_parts(text, &boundaries, config)
}

fn parse_keywords(raw: &str) -> Vec<String> {
    raw.split(',').map(|k| k.trim().to_lowercase()).filter(|k| !k.is_empty()).collect()
}

fn is_all_caps(line: &str) -> bool {
    let mut has_cased = false;
    for c in line.chars() {
        if c.is_lowercase() {
            return false;
        }
        if c.is_uppercase() {
            has_cased = true;
        }
    }
    has_cased
}

fn collect_boundaries(text: &str, headings: &[BookHeading], config: &SplitConfig) -> Vec<Boundary> {
    let mut boundaries: Vec<Boundary> = Vec::new();
    let keywords = parse_keywords(&config.keywords);
    let mut blank_run = 0usize;
    let mut offset = 0usize;
    for line in text.split_inclusive('\n') {
        let line_body = line.trim_end_matches(['\n', '\r']);
        let trimmed = line_body.trim();
        if trimmed.is_empty() {
            blank_run += 1;
        } else {
            let line_start = offset + (line_body.len() - line_body.trim_start().len());
            if config.by_blank_lines && blank_run >= 2 {
                boundaries.push(Boundary { offset: line_start, title: None, rank: 0 });
            }
            let lower = trimmed.to_lowercase();
            if config.by_keywords && keywords.iter().any(|k| lower.starts_with(k.as_str())) {
                boundaries.push(Boundary { offset: line_start, title: Some(trimmed.to_string()), rank: 1 });
            }
            if config.by_all_caps && is_all_caps(trimmed) {
                boundaries.push(Boundary { offset: line_start, title: Some(trimmed.to_string()), rank: 1 });
            }
            blank_run = 0;
        }
        offset += line.len();
    }
    if config.by_toc {
        for heading in headings {
            if heading.offset <= text.len() {
                boundaries.push(Boundary { offset: heading.offset, title: Some(heading.title.clone()), rank: 2 });
            }
        }
    }
    boundaries.sort_by(|a, b| a.offset.cmp(&b.offset).then(b.rank.cmp(&a.rank)));
    boundaries.dedup_by_key(|b| b.offset);
    boundaries
}

fn part_text(piece: &str, strip_heading: bool) -> &str {
    let trimmed = piece.trim();
    if strip_heading {
        trimmed.split_once('\n').map(|(_, rest)| rest.trim()).unwrap_or("")
    } else {
        trimmed
    }
}

fn form_parts(text: &str, boundaries: &[Boundary], config: &SplitConfig) -> Vec<BookPart> {
    let mut parts: Vec<BookPart> = Vec::new();
    let mut start = 0usize;
    let mut current_title: Option<String> = None;
    let mut strip_heading = false;
    for boundary in boundaries {
        if boundary.offset == 0 {
            current_title = boundary.title.clone();
            strip_heading = boundary.title.is_some();
            continue;
        }
        if boundary.offset <= start {
            continue;
        }
        let candidate = part_text(&text[start..boundary.offset], strip_heading && config.remove_heading);
        if !candidate.is_empty() {
            parts.push(BookPart { title: current_title.take(), text: candidate.to_string() });
        }
        start = boundary.offset;
        current_title = boundary.title.clone();
        strip_heading = boundary.title.is_some();
    }
    let tail = part_text(&text[start..], strip_heading && config.remove_heading);
    if !tail.is_empty() {
        parts.push(BookPart { title: current_title, text: tail.to_string() });
    }
    parts
}
```

```rust
fn apply_size_cap(parts: Vec<BookPart>, max_chars: usize) -> Vec<BookPart> {
    let max_chars = max_chars.max(1);
    let mut out = Vec::new();
    for part in parts {
        if part.text.chars().count() <= max_chars {
            out.push(part);
            continue;
        }
        for (index, sub) in split_text_by_size(&part.text, max_chars).into_iter().enumerate() {
            out.push(BookPart {
                title: if index == 0 { part.title.clone() } else { None },
                text: sub,
            });
        }
    }
    out
}

fn split_text_by_size(text: &str, max_chars: usize) -> Vec<String> {
    use crate::services::chunker::{split_into_sentences, split_long_sentence};
    let mut chunks: Vec<String> = Vec::new();
    let mut current = String::new();
    for line in text.lines() {
        let line = line.trim();
        if line.is_empty() {
            continue;
        }
        if line.chars().count() > max_chars {
            if !current.is_empty() {
                chunks.push(std::mem::take(&mut current));
            }
            for sentence in split_into_sentences(line) {
                if sentence.chars().count() > max_chars {
                    if !current.is_empty() {
                        chunks.push(std::mem::take(&mut current));
                    }
                    chunks.extend(split_long_sentence(&sentence, max_chars));
                } else {
                    push_line(&mut chunks, &mut current, &sentence, max_chars);
                }
            }
            continue;
        }
        push_line(&mut chunks, &mut current, line, max_chars);
    }
    if !current.is_empty() {
        chunks.push(current);
    }
    chunks
}

fn push_line(chunks: &mut Vec<String>, current: &mut String, line: &str, max_chars: usize) {
    let sep = if current.is_empty() { 0 } else { 1 };
    if !current.is_empty() && current.chars().count() + sep + line.chars().count() > max_chars {
        chunks.push(std::mem::take(current));
    }
    if !current.is_empty() {
        current.push('\n');
    }
    current.push_str(line);
}
```

```rust
pub fn part_file_name(config: &SplitConfig, part: &BookPart, index: usize, total: usize, source_name: &str) -> String {
    let number = config.start_number + index;
    let width = (config.start_number + total.saturating_sub(1)).to_string().len().max(2);
    let padded = format!("{number:0width$}");
    let first_line: String = part
        .text
        .lines()
        .map(str::trim)
        .find(|line| !line.is_empty())
        .unwrap_or("")
        .chars()
        .take(60)
        .collect();
    let header = part.title.as_deref().unwrap_or(&first_line);
    let template = if config.template.trim().is_empty() {
        SplitConfig::default().template
    } else {
        config.template.clone()
    };
    let mut name = template
        .replace("%Number%", &padded)
        .replace("%FileName%", source_name)
        .replace("%FirstLine%", &first_line)
        .replace("%Header%", header);
    if config.add_number && !template.contains("%Number%") {
        name.push_str(&format!(" {padded}"));
    }
    sanitize_file_name(&name)
}

pub fn sanitize_file_name(name: &str) -> String {
    let mut out = String::new();
    let mut last_space = false;
    for c in name.chars() {
        let c = if matches!(c, '/' | '\\' | '?' | '*' | ':' | '"' | '<' | '>' | '|') || c.is_control() {
            '_'
        } else {
            c
        };
        if c.is_whitespace() {
            if !last_space && !out.is_empty() {
                out.push(' ');
            }
            last_space = true;
        } else {
            out.push(c);
            last_space = false;
        }
    }
    let trimmed = out.trim().trim_end_matches(['.', ' ']);
    let capped: String = trimmed.chars().take(120).collect();
    let capped = capped.trim_end_matches(['.', ' ']).to_string();
    if capped.is_empty() { "часть".to_string() } else { capped }
}
```

```rust
use std::collections::HashSet;
use std::path::PathBuf;

#[derive(Debug, Default)]
pub struct ExportSummary {
    pub saved: usize,
    pub dir: PathBuf,
    pub errors: Vec<String>,
}

pub fn export_parts(parts: &[BookPart], config: &SplitConfig, source_name: &str) -> Result<ExportSummary, String> {
    let base = config.output_dir.as_ref().ok_or("Не выбрана папка назначения")?;
    let dir = if config.subfolder {
        base.join(sanitize_file_name(source_name))
    } else {
        base.clone()
    };
    std::fs::create_dir_all(&dir)
        .map_err(|e| format!("Не удалось создать папку {}: {e}", dir.display()))?;
    let mut used: HashSet<String> = HashSet::new();
    let mut summary = ExportSummary { dir: dir.clone(), ..Default::default() };
    for (index, part) in parts.iter().enumerate() {
        let name = part_file_name(config, part, index, parts.len(), source_name);
        let target = crate::services::batch::unique_target(
            &dir,
            &format!("{name}.txt"),
            config.overwrite,
            &used,
        );
        used.insert(crate::services::batch::target_key(&target));
        match std::fs::write(&target, &part.text) {
            Ok(()) => summary.saved += 1,
            Err(e) => summary.errors.push(format!("{}: {e}", target.display())),
        }
    }
    Ok(summary)
}
```

```rust
fn extract_xml_paragraphs(
    xml_str: &str,
    block_tags: &[&str],
    skip_tags: &[&str],
    cancel: Option<&AtomicBool>,
    mut headings: Option<&mut Vec<BookHeading>>,
) -> Result<String, String> {
    let mut reader = Reader::from_str(xml_str);
    reader.config_mut().trim_text(false);

    let mut out = String::with_capacity(xml_str.len() / 2);
    let mut skip_depth = 0usize;
    let mut in_body = !xml_str.contains("<body");
    let mut section_depth = 0usize;
    let mut title_start: Option<usize> = None;

    loop {
        if book_cancelled(cancel) {
            return Err("Загрузка книги отменена".to_string());
        }
        match reader.read_event() {
            Ok(Event::Start(e)) => {
                let local_bytes = e.local_name();
                let local = std::str::from_utf8(local_bytes.as_ref())
                    .unwrap_or("")
                    .to_ascii_lowercase();
                if local == "body" {
                    in_body = true;
                }
                if skip_tags.iter().any(|&t| t == local) {
                    skip_depth += 1;
                } else if skip_depth == 0
                    && in_body
                    && block_tags.iter().any(|&t| t == local)
                    && !out.is_empty()
                    && !out.ends_with('\n')
                {
                    out.push('\n');
                }
                if skip_depth == 0 && in_body {
                    if local == "section" {
                        section_depth += 1;
                    }
                    if local == "title" && section_depth > 0 && headings.is_some() {
                        title_start = Some(out.len());
                    }
                }
            }
            Ok(Event::End(e)) => {
                let local_bytes = e.local_name();
                let local = std::str::from_utf8(local_bytes.as_ref())
                    .unwrap_or("")
                    .to_ascii_lowercase();
                if skip_tags.iter().any(|&t| t == local) {
                    skip_depth = skip_depth.saturating_sub(1);
                } else if skip_depth == 0
                    && in_body
                    && block_tags.iter().any(|&t| t == local)
                    && !out.ends_with('\n')
                {
                    out.push('\n');
                }
                if skip_depth == 0 && in_body {
                    if local == "section" {
                        section_depth = section_depth.saturating_sub(1);
                    }
                    if local == "title" {
                        if let (Some(start), Some(headings)) = (title_start.take(), headings.as_mut()) {
                            if let Some(title) = out[start..].lines().map(str::trim).find(|l| !l.is_empty()) {
                                headings.push(BookHeading { offset: start, title: title.to_string() });
                            }
                        }
                    }
                }
            }
            Ok(Event::Text(t)) if skip_depth == 0 && in_body => {
                match t.unescape_with(resolve_html_entity) {
                    Ok(unescaped) => out.push_str(&unescaped),
                    Err(_) => out.push_str(&String::from_utf8_lossy(&t)),
                }
            }
            Ok(Event::Eof) => break,
            Err(e) => return Err(format!("Ошибка разбора XML: {e}")),
            _ => {}
        }
    }

    let trim_start = out.len() - out.trim_start().len();
    if let Some(headings) = headings.as_mut() {
        for heading in headings.iter_mut() {
            heading.offset = heading.offset.saturating_sub(trim_start);
        }
    }
    Ok(out.trim().to_string())
}
```
